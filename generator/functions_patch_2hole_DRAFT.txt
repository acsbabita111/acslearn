/* ══ (30-Jul होल-1, Founder-override v4.0-ख1) अभिरुचि-नतीजा खाते में ══
   client: apt-pay.js v1.1 → saveResult / latestResult।
   एक uid = एक doc (aptitudeResults/{uid}) — हर final-submit पर ताज़ा। */
function _aptClean(t, max) {
  return String(t == null ? "" : t).replace(/[<>]/g, "").slice(0, max);
}
exports.saveAptitudeResult = functions.region(REGION).https.onCall(async (data, context) => {
  try {
    if (!context.auth) throw new functions.https.HttpsError("unauthenticated", "पहले login करें।");
    const uid = context.auth.uid;
    const p = (data && data.prev) || {};
    const mg = Array.isArray(p.mg) ? p.mg.slice(0, 6).map(x => _aptClean(x, 60)) : [];
    const udy = Array.isArray(p.udy) ? p.udy.slice(0, 10).map(u => ({
      n: Math.max(0, Math.min(5000, parseInt(u && u.n, 10) || 0)),
      nm: _aptClean(u && u.nm, 120),
      c: (function () { const c = _aptClean(u && u.c, 200); return c.charAt(0) === "/" ? c : ""; })(),
      s: Number(u && u.s) || 0
    })) : [];
    if (!mg.length && !udy.length)
      throw new functions.https.HttpsError("invalid-argument", "नतीजा ख़ाली है।");
    await db.collection("aptitudeResults").doc(uid).set({
      uid: uid,
      at: Date.now(),                       /* client-समय नहीं — server-घड़ी */
      updatedAt: admin.firestore.FieldValue.serverTimestamp(),
      prev: { at: Date.now(), mg: mg, udy: udy },
      count: admin.firestore.FieldValue.increment(1)   /* कुल प्रयास-गिनती */
    }, { merge: true });
    return { ok: true };
  } catch (e) {
    if (e instanceof functions.https.HttpsError) throw e;
    console.error("saveAptitudeResult:", e);
    throw new functions.https.HttpsError("internal", "saveAptitudeResult: " + (e && e.message));
  }
});
exports.latestAptitudeResult = functions.region(REGION).https.onCall(async (data, context) => {
  try {
    if (!context.auth) throw new functions.https.HttpsError("unauthenticated", "पहले login करें।");
    const doc = await db.collection("aptitudeResults").doc(context.auth.uid).get();
    if (!doc.exists) return { found: false };
    const d = doc.data() || {};
    return { found: true, prev: d.prev || null, at: d.at || null, count: d.count || 0 };
  } catch (e) {
    if (e instanceof functions.https.HttpsError) throw e;
    console.error("latestAptitudeResult:", e);
    throw new functions.https.HttpsError("internal", "latestAptitudeResult: " + (e && e.message));
  }
});
